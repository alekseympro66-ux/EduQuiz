        function renderQuestions() {
            questionsContainer.innerHTML = '';

            if (!state.currentCategory) return;

            state.currentCategory.questions.forEach((question, index) => {
                const questionKey = `${state.currentCategory.id}-${question.id}`;
                const isAnswered = state.currentUser.answeredQuestions[questionKey];
                const isCorrect = isAnswered === true;

                const questionCard = document.createElement('div');
                questionCard.className = `question-card ${isAnswered ? (isCorrect ? 'answered' : 'incorrect') : ''}`;
                questionCard.innerHTML = `
                    <div class="question-number">Вопрос ${index + 1}</div>
                    <div class="question-text">${question.question}</div>
                    <div class="question-points">
                        <div class="points-badge">${question.points} балл</div>
                        <div class="question-status ${isCorrect ? 'correct' : 'incorrect'}">
                            ${isAnswered ? (isCorrect ? '✓' : '✗') : ''}
                        </div>
                    </div>
                `;

                questionCard.addEventListener('click', () => startQuestion(question));

                questionsContainer.appendChild(questionCard);
            });
        }

        function startQuestion(question) {
            state.currentQuestion = question;
            quizCategoryTitle.textContent = state.currentCategory.name;
            questionText.textContent = question.question;
            textAnswer.value = '';
            textAnswer.disabled = false;

            // Clear feedback
            answerFeedback.className = 'answer-feedback';
            answerFeedback.textContent = '';

            showScreen('quizScreen');
        }

        function checkAnswer() {
            const userAnswer = textAnswer.value.trim().toLowerCase();
            const correctAnswer = state.currentQuestion.answer.toLowerCase();

            const isCorrect = userAnswer === correctAnswer;
            const questionKey = `${state.currentCategory.id}-${state.currentQuestion.id}`;

            // Only count if not already answered
            if (!state.currentUser.answeredQuestions[questionKey]) {
                state.currentUser.answeredQuestions[questionKey] = isCorrect;

                if (isCorrect) {
                    state.currentUser.points += state.currentQuestion.points;
                    state.currentUser.correctAnswers++;
                    answerFeedback.className = 'answer-feedback correct';
                    answerFeedback.textContent = 'Правильно!';
                    playSound(correctSound);
                } else {
                    answerFeedback.className = 'answer-feedback incorrect';
                    answerFeedback.textContent = 'Неправильно! Попробуйте еще раз.';
                    playSound(incorrectSound);
                }

                // Check if user unlocked new categories
                let unlockedNew = false;
                [...state.categories1, ...state.categories2].forEach(category => {
                    if (category.locked &&
                        !state.currentUser.unlockedCategories.includes(category.id) &&
                        state.currentUser.points >= category.requiredPoints) {
                        state.currentUser.unlockedCategories.push(category.id);
                        unlockedNew = true;
                    }
                });

                localStorage.setItem('eduquiz_user', JSON.stringify(state.currentUser));
                updateProfileInfo();

                if (unlockedNew) {
                    playSound(unlockSound);
                    setTimeout(() => {
                        alert(`Поздравляем! Вы открыли новые категории!`);
                        renderCategories();
                    }, 1000);
                }
            } else {
                // Already answered, just show feedback
                if (isCorrect) {
                    answerFeedback.className = 'answer-feedback correct';
                    answerFeedback.textContent = 'Правильно!';
                } else {
                    answerFeedback.className = 'answer-feedback incorrect';
                    answerFeedback.textContent = 'Неправильно! Попробуйте еще раз.';
                }
            }

            textAnswer.disabled = true;

            // Update the questions view
            setTimeout(() => {
                renderQuestions();
            }, 1500);
        }

        // Profile
        function updateProfileInfo() {
            if (!state.currentUser) return;

            profileUsername.textContent = state.currentUser.username;
            profileJoinDate.textContent = state.currentUser.joinDate;
            totalPoints.textContent = state.currentUser.points;
            completedQuizzes.textContent = Object.keys(state.currentUser.answeredQuestions).length;
            correctAnswers.textContent = state.currentUser.correctAnswers;
        }

        // Settings
        function handleThemeChange(e) {
            const theme = e.target.getAttribute('data-theme');

            // Update active theme button
            themeOptions.forEach(option => {
                option.classList.remove('active');
            });
            e.target.classList.add('active');

            // Apply theme to body
            document.body.className = '';
            if (theme !== 'default') {
                document.body.classList.add(`${theme}-theme`);
            }
        }

        function handleMusicVolumeChange() {
            backgroundMusic.volume = musicVolume.value / 100;
        }

        function handleSoundVolumeChange() {
            buttonSound.volume = soundVolume.value / 100;
            correctSound.volume = soundVolume.value / 100;
            incorrectSound.volume = soundVolume.value / 100;
            unlockSound.volume = soundVolume.value / 100;
        }

        function handleDeleteAccount() {
            if (state.currentUser) {
                localStorage.removeItem('eduquiz_user');
                state.currentUser = null;
                hideModal('deleteAccountModal');
                authScreen.style.display = 'flex';
                app.style.display = 'none';
            }
        }

        // Utility functions
        function showModal(modalId) {
            document.getElementById(modalId).style.display = 'flex';
        }

        function hideModal(modalId) {
            document.getElementById(modalId).style.display = 'none';
        }

        function playSound(sound) {
            sound.currentTime = 0;
            sound.play().catch(e => console.log("Sound play prevented:", e));
        }

        // Initialize the app when the page loads
        window.addEventListener('DOMContentLoaded', init);
    </script>
